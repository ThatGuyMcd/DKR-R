# Optional private-fixture qualification: consume Patch Pipeline output, never
# alter it. Only asset APIs, DMA functions and original list/free helpers link.
set(DKR_LEGACY_PIPELINE_ROOT "" CACHE PATH "Directory containing regenerated RecompiledFuncs-v77/v80")
if(NOT DKR_LEGACY_PIPELINE_ROOT STREQUAL "")
    foreach(rev IN ITEMS 77 80)
        set(generated "${DKR_LEGACY_PIPELINE_ROOT}/RecompiledFuncs-v${rev}")
        set(copy_sources "${generated}/dmacopy.c")
        if(rev EQUAL 80)
            list(APPEND copy_sources "${generated}/dmacopy_v1.c")
        endif()
        set(target DKRLegacyGuestPipelineV${rev})
        add_executable(${target} ../legacy_guest_pipeline_tests.cpp
            ${copy_sources} "${generated}/__amDMA.c"
            "${generated}/alLink.c" "${generated}/alUnlink.c" "${generated}/tex_free.c" "${generated}/alSeqFileNew.c"
            "${generated}/asset_table_load.c" "${generated}/asset_load.c"
            "${generated}/asset_rom_offset.c" "${generated}/asset_table_size.c"
            "${generated}/asset_table_load_addr.c" "${generated}/asset_table_load_zipped.c")
        target_link_libraries(${target} PRIVATE DKRLegacyGuestIO)
        target_include_directories(${target} PRIVATE "${generated}")
        target_compile_definitions(${target} PRIVATE DKR_TEST_REVISION=${rev})
        # The recompiler's one-function-per-file mode omits funcs.h includes.
        # Apply its own generated declarations through the build boundary.
        if(MSVC)
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:/FIfuncs.h>")
            # Runtime guest callbacks can unwind through regenerated C. Do not
            # assume C-linkage callbacks are nonthrowing in this fault harness.
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/EHs;/EHc->")
        else()
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-include;funcs.h>")
            # Regenerated MIPS LUI uses signed 32-bit shifts deliberately. Keep
            # ASan and other UBSan checks, but not signed shift-base checks in
            # generated C. New project-owned C++ retains the full sanitizer set.
            target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-fno-sanitize=shift-base>")
        endif()
    endforeach()
endif()
