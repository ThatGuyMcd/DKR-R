cmake_minimum_required(VERSION 3.24)
project(DKRTestAssertionContract LANGUAGES C CXX)
include(CTest)
add_executable(DKRBeforeIncludeTests assertions.c)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/ProjectTests.cmake")
add_executable(DKRAfterIncludeTests assertions.cpp)
# A target-level definition must not accidentally override the test contract.
target_compile_definitions(DKRAfterIncludeTests PRIVATE NDEBUG)
add_executable(DKRProduction production.cpp)
add_subdirectory(vendor)
add_test(NAME before COMMAND DKRBeforeIncludeTests)
add_test(NAME after COMMAND DKRAfterIncludeTests)
add_test(NAME production COMMAND DKRProduction)
add_test(NAME vendor COMMAND DKRVendorTests)
