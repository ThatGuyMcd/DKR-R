if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/native/party.cpp")
 message(FATAL_ERROR "Configure the isolated output of stage_core.py, not the authored directory")
endif()
add_library(DKRSpectatorPolicy STATIC native/party.cpp native/confirmed_feed.cpp native/party_codec.cpp native/checkpoint_transfer.cpp native/observer_playback.cpp native/confirmed_observer.cpp native/owner_export.cpp native/owned_observer.cpp)
target_compile_features(DKRSpectatorPolicy PUBLIC cxx_std_20)

if(MSVC)
 target_compile_options(DKRSpectatorPolicy PRIVATE /W4 /WX /permissive- /EHsc)
else()
 target_compile_options(DKRSpectatorPolicy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
