cmake_minimum_required(VERSION 3.20)
project(DKRStabilityPolicyChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(DKRStabilityPolicyChecks policy_tests.cpp)
target_include_directories(DKRStabilityPolicyChecks PRIVATE ../native)
target_link_libraries(DKRStabilityPolicyChecks PRIVATE Threads::Threads)
enable_testing()
add_test(NAME DKRStabilityPolicyChecks COMMAND DKRStabilityPolicyChecks)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/sp_fixture.cpp" fixture)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sdk_task_source.hpp" "#pragma once\n#define DKR_STABILITY_ORIGINAL_SP \"${fixture}\"\n")
add_executable(DKRStabilityTaskSubmissionChecks task_submission_tests.cpp ../native/task_submission.cpp)
target_include_directories(DKRStabilityTaskSubmissionChecks PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ../native)
target_link_libraries(DKRStabilityTaskSubmissionChecks PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(DKRStabilityTaskSubmissionChecks PRIVATE /EHs /EHc-)
endif()
add_test(NAME DKRStabilityTaskSubmissionChecks COMMAND DKRStabilityTaskSubmissionChecks)
